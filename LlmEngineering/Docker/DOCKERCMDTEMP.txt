# Build a custom image (recommended for reproducibility)
# =========================
python --version
pip --version

# =========================
# Method 1: Use requirements text files
# =========================
# Dockerfile
vi Dockerfile
# Contents:
FROM jupyter/scipy-notebook
COPY np_pkg.txt statemodel_pkg.txt viz_pkg.txt ml_pkg.txt ./
RUN pip install -r np_pkg.txt \
    -r statemodel_pkg.txt \
    -r viz_pkg.txt \
    -r ml_pkg.txt

# Build image
docker build -t mlx_pkg .

# =========================
# Method 2: Direct install in Dockerfile
# =========================
vi Dockerfile
# Contents:
FROM jupyter/scipy-notebook
RUN pip install plotly sympy arviz scikit-learn pandas matplotlib seaborn

# Build image
docker build -t mlx_pkg .
# If you want a clean slate, delete the old image
docker rmi mlx_pkg
# To see the list of Python packages inside your custom image
docker run -it --rm mlx_pkg pip list
